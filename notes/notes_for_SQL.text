create database (if not exists) test; // creat a database

show databases; // show all the databases

create database itheima default charset utf8mb4; // 指定其字符集为四个字节

drop database if exists test; // delete a database


use itcast; // use database "itcast"
select database(); // check which database is being used

show tables; // check all tables (must use one database)

desc name_table; // 查看表结构
show create table name_table; // 查看建表语句

create table name_table(
    字段1 字段1类型[comment for the 字段], // no[]
    ...
)[comment for table];
e.g.

create table id(
    -> id int comment "编号",
    -> name varchar(20),
    -> age int,
    -> gender varchar(1) // 最后一个无逗号
    -> )comment '用户表';
