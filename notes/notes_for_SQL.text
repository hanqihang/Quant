show tables; // check all tables (must use one database)

desc name_table; // 查看表结构
show create table name_table; // 查看建表语句

create table name_table(
    字段1 字段1类型[comment for the 字段], // no[]
    ...
)[comment for table];
e.g.

create table id(
    id int comment "编号",
    name varchar(20),
    age int,
    gender varchar(1) // 最后一个无逗号
    )comment '用户表';

select * from mysql.user; // 查看所有在mysql.user中的数据

create table t2 select host,user from mysql.user where 1 > 5; // 仅复制表结构
